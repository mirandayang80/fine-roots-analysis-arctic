# make first row the header
FRED3 <- FRED3 |> row_to_names(1)

# remove irrelevant rows
FRED3 <- FRED3 |>
  filter(Name == "1" | Name == "Type")

# pulling the first row as a vector 
var_type <- unlist(FRED3[1,], use.names = FALSE)

# get vector of indices that correspond to root traits
indices <- rep(0, length(var_type))

# looping through entire row of variable types
for (i in 1:length(var_type)) {
  
  # if entry is empty
  if(is.na(var_type[i])){
    # skip that entry
    i = i + 1
  }
  # if the variable type is root trait
  else if (var_type[i] == "Root_Traits") {
   
    # add the index to our vector
    indices[i] <- i

  }
}

# dropping "0" values from indices vector
indices <- indices[indices != 0]

# dplyr::selecting columns with article, lat, long, and root traits
FRED3_cleaned <- FRED3 |>
  # create short variable name for article
  rename(Article = "Abbreviated article citation", 
         order = "Plant taxonomy_Accepted order_APW",
         family = "Plant taxonomy_Accepted family_TPL", 
         genus = "Plant taxonomy_Accepted genus_TPL", 
         species = "Plant Taxonomy_Accepted species_TPL") |>
  # select article, coordinates, and root trait columns
  dplyr::select(Article, Latitude_Main, Longitude_Main,
                order, family, genus, species, indices)

FRED3_cleaned <- FRED3_cleaned[3:nrow(FRED3_cleaned),]

# aggregate data by Year and Category
yearly_trends <- FRED4arctic_with_countries |>
  # Ensure we only count each article once 
  distinct(Article, .keep_all = TRUE) |>
  # Clean up and extract the 4-digit year from the citation
  mutate(Article = str_replace(Article,
                          "Shaver & Billlings 197$",
                          "Shaver & Billlings 1975"),
    Year = as.numeric(str_extract(Article, "[0-9]{4}"))
  ) |>
  group_by(Year) |>
  summarise(Paper_Count = n(), .groups = "drop")

# plot the timeline trend
ggplot(yearly_trends, aes(x = Year, y = Paper_Count)) +
  # Move fill directly inside geom_col to apply it uniformly to all bars
  geom_col(width = 1, fill = "#00A5CF", color = "black", linewidth = 0.2) + 
  
  scale_x_continuous(breaks = seq(1920, 2025, by = 10)) +
  scale_y_continuous(
    # Skips by 2
    breaks = seq(0, max(yearly_trends$Paper_Count, na.rm = TRUE) + 2, by = 2)
  ) + 
  theme_minimal() +
  labs(
    x = "Publication Year",
    y = "Number of Papers" 
  ) +
  theme(
    text = element_text(size = 11)
  )
decade_trends <- yearly_trends |>

  summarise(Paper_Count = sum(Paper_Count), .groups = "drop")
